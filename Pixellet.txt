<script>

let currentUser = "";
let coins = 0;
let lastClaim = 0;

function showError(message){

  errorBubble.innerHTML = message;

  errorBubble.style.display = "block";

  setTimeout(() => {
    errorBubble.style.display = "none";
  },3000);
}

function signup(){

  let usernameValue =
    document.getElementById("username").value;

  let passwordValue =
    document.getElementById("password").value;

  let users =
    JSON.parse(localStorage.getItem("users")) || {};

  if(localStorage.getItem("existingPixelletAccount")){

    showError(
      "You already created a Pixellet account on this device."
    );

    return;
  }

  if(users[usernameValue]){

    showError(
      "This username already exists."
    );

    return;
  }

  users[usernameValue] =
    passwordValue;

  localStorage.setItem(
    "users",
    JSON.stringify(users)
  );

  localStorage.setItem(
    "existingPixelletAccount",
    usernameValue
  );

  alert("Account created!");

  login();
}

function login(){

  let usernameValue =
    document.getElementById("username").value;

  let passwordValue =
    document.getElementById("password").value;

  let users =
    JSON.parse(localStorage.getItem("users")) || {};

  if(
    users[usernameValue] &&
    users[usernameValue] === passwordValue
  ){

    currentUser = usernameValue;

    coins =
      Number(
        localStorage.getItem(
          "coins_" + currentUser
        )
      ) || 0;

    lastClaim =
      Number(
        localStorage.getItem(
          "lastClaim_" + currentUser
        )
      ) || 0;

    coinAmount.innerHTML = coins;

    loginPage.style.display = "none";
    profilePage.style.display = "block";
    sidebar.style.display = "block";

    topUsername.innerHTML =
      currentUser;

    let badge =
      document.getElementById("userBadge");

    if(currentUser === "OGsacredYT"){

      badge.innerHTML =
        "?? OWNER";

      badge.classList.add(
        "owner-badge"
      );

    }else{

      badge.innerHTML =
        "NEW";

      badge.classList.remove(
        "owner-badge"
      );
    }

  }else{

    showError(
      "Wrong username or password."
    );
  }
}

function showHome(){

  profilePage.style.display = "block";
  packsPage.style.display = "none";
  chatPage.style.display = "none";
}

function showPacks(){

  profilePage.style.display = "none";
  packsPage.style.display = "flex";
  chatPage.style.display = "none";
}

function showChat(){

  profilePage.style.display = "none";
  packsPage.style.display = "none";
  chatPage.style.display = "flex";

  chatMessages.innerHTML =
    localStorage.getItem("pixelletChat") || "";
}

function sendMessage(){

  let input =
    document.getElementById("chatInput");

  if(input.value === "") return;

  chatMessages.innerHTML +=
    "<p><b>" +
    currentUser +
    ":</b> " +
    input.value +
    "</p>";

  localStorage.setItem(
    "pixelletChat",
    chatMessages.innerHTML
  );

  input.value = "";
}

chatInput.addEventListener(
  "keydown",
  function(e){

    if(e.key === "Enter"){

      sendMessage();
    }

});

function hourlyReward(){

  let now = Date.now();

  if(now - lastClaim < 3600000){

    alert("Already claimed.");

    return;
  }

  lastClaim = now;

  localStorage.setItem(
    "lastClaim_" + currentUser,
    lastClaim
  );

  let reward =
    Math.floor(Math.random() * 2001) + 500;

  coins += reward;

  localStorage.setItem(
    "coins_" + currentUser,
    coins
  );

  coinAmount.innerHTML =
    coins;

  /* REWARD SCREEN */

  let rewardScreen =
    document.createElement("div");

  rewardScreen.style.position = "fixed";
  rewardScreen.style.top = "0";
  rewardScreen.style.left = "0";
  rewardScreen.style.width = "100%";
  rewardScreen.style.height = "100%";
  rewardScreen.style.background =
    "rgba(0,0,0,.88)";
  rewardScreen.style.zIndex = "99999";
  rewardScreen.style.overflow = "hidden";

  document.body.appendChild(
    rewardScreen
  );

  let rewardText =
    document.createElement("div");

  rewardText.innerHTML =
    "+" + reward + " COINS!";

  rewardText.style.position =
    "absolute";

  rewardText.style.top = "50%";
  rewardText.style.left = "50%";

  rewardText.style.transform =
    "translate(-50%,-50%)";

  rewardText.style.fontSize =
    "70px";

  rewardText.style.fontWeight =
    "900";

  rewardText.style.color =
    "#00ccff";

  rewardText.style.webkitTextStroke =
    "3px white";

  rewardScreen.appendChild(
    rewardText
  );

  for(let i = 0; i < 120; i++){

    let confetti =
      document.createElement("div");

    confetti.style.position =
      "absolute";

    confetti.style.width =
      "12px";

    confetti.style.height =
      "28px";

    confetti.style.left =
      Math.random() * 100 + "%";

    confetti.style.top =
      "-50px";

    confetti.style.background =
      i % 2 === 0
      ? "#00ccff"
      : "#0044ff";

    confetti.style.transform =
      "rotate(" +
      Math.random() * 360 +
      "deg)";

    confetti.style.animation =
      "confettiFall " +
      (Math.random() * 3 + 2) +
      "s linear forwards";

    rewardScreen.appendChild(
      confetti
    );
  }

  setTimeout(() => {

    rewardScreen.remove();

  },4000);
}

setInterval(() => {

  let left =
    3600000 - (Date.now() - lastClaim);

  if(left <= 0){

    timerText.innerHTML =
      "Ready!";

  }else{

    let m =
      Math.floor(left / 60000);

    let s =
      Math.floor((left % 60000) / 1000);

    timerText.innerHTML =
      m + "m " + s + "s";
  }

},1000);
